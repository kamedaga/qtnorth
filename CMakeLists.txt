cmake_minimum_required(VERSION 3.21)
project(qtNorth VERSION 0.1.2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
    add_compile_options(/utf-8 /permissive-)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Svg)
find_package(Qt6Keychain REQUIRED)
qt_standard_project_setup()

add_library(north_core src/northapi.cpp src/post.cpp src/postformat.cpp)
target_include_directories(north_core PUBLIC src)
target_link_libraries(north_core PUBLIC Qt6::Network)

add_library(north_ui src/mainwindow.cpp src/postcard.cpp src/icons.cpp src/remoteimage.cpp src/messagespage.cpp src/scrollbehavior.cpp src/loginpage.cpp src/credentialstore.cpp src/theme.cpp src/aboutdialog.cpp src/profilepage.cpp src/posttime.cpp src/windowchrome.cpp src/notificationspage.cpp)
target_compile_definitions(north_ui PUBLIC QTNORTH_VERSION="${PROJECT_VERSION}")
target_sources(north_ui PRIVATE src/settingsdialog.cpp src/preferences.cpp)
target_sources(north_ui PRIVATE src/posttext.cpp src/linkpreview.cpp)
target_sources(north_ui PRIVATE src/reactionusers.cpp)
target_sources(north_ui PRIVATE src/pollwidget.cpp src/userrelations.cpp src/postcomposer.cpp)
target_link_libraries(north_ui PUBLIC north_core Qt6::Widgets Qt6::Svg Qt6Keychain::Qt6Keychain)
qt_add_executable(qtNorth WIN32 src/main.cpp)
target_link_libraries(qtNorth PRIVATE north_ui)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT CMAKE_SKIP_RPATH
   AND EXISTS "${CMAKE_CURRENT_BINARY_DIR}/runtime/lib"
   AND EXISTS "${CMAKE_CURRENT_BINARY_DIR}/runtime/plugins")
    # The extracted SDK is not installed in the system loader's search paths.
    # DT_RPATH also covers indirect Qt dependencies such as DBus and libb2.
    set_property(TARGET qtNorth APPEND PROPERTY BUILD_RPATH "$ORIGIN/runtime/lib")
    target_link_options(qtNorth PRIVATE "LINKER:--disable-new-dtags")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/qt.conf"
        CONTENT "[Paths]\nPrefix=runtime\nLibraries=lib\nPlugins=plugins\n")
endif()
if(WIN32)
    enable_language(RC)
    configure_file(assets/qtNorth.rc.in ${CMAKE_CURRENT_BINARY_DIR}/qtNorth.rc @ONLY)
    target_sources(qtNorth PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/qtNorth.rc)
endif()
qt_add_resources(qtNorth assets PREFIX "/" FILES
    assets/qtNorth-icon.svg assets/qtNorth-lockup-dark.svg assets/qtNorth-lockup-light.svg scripts/windows-credentials.ps1 DISCLAIMER.txt)
install(TARGETS qtNorth BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES DISCLAIMER.txt DESTINATION share/doc/qtnorth)
if(UNIX AND NOT APPLE)
    install(FILES packaging/rip.north.qtNorth.desktop DESTINATION share/applications)
    install(FILES assets/qtNorth-icon.svg DESTINATION share/icons/hicolor/scalable/apps RENAME qtnorth.svg)
endif()
